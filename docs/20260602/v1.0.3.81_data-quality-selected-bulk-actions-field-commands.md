## 驗證

只輸出版本與統計，不輸出主機、IP、資產編號或人名。

```bash
python3 - <<'PY'
import json
import urllib.request

BASE = "http://127.0.0.1:8002"

def get_json(path):
    with urllib.request.urlopen(BASE + path, timeout=15) as res:
        return json.loads(res.read().decode("utf-8"))

health = get_json("/health")
quality = get_json("/api/reports/data-quality")
summary = quality.get("summary", {})

print(json.dumps({
    "version": health.get("version"),
    "patch_id": health.get("patch_id"),
    "health_status": health.get("status"),
    "quality_status": quality.get("status"),
    "platform_mismatch_count": summary.get("cmdb_platform_mismatch"),
    "missing_connection_count": summary.get("cmdb_missing_connection"),
    "workflow_pending_count": summary.get("cmdb_workflow_pending")
}, ensure_ascii=False, indent=2, sort_keys=True))
PY
```

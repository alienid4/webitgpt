import os
import re
import urllib.request

base = os.environ.get("BASE_URL", "http://127.0.0.1:8002").rstrip("/")

def get(path):
    with urllib.request.urlopen(base + path, timeout=10) as response:
        return response.status, response.read().decode("utf-8", errors="replace")

health_status, health = get("/health")
hosts_status, hosts = get("/hosts")

checks = {
    "health_http_ok": health_status == 200,
    "hosts_http_ok": hosts_status == 200,
    "version_1_0_3_74": '"version":"1.0.3.74"' in health.replace(" ", ""),
    "bulk_status_form": "bulkStatusForm" in hosts,
    "select_assets": "data-select-assets" in hosts,
    "include_inactive_toggle": "include_inactive" in hosts,
    "default_inactive_hint": "顯示下線 / 待刪除" in hosts,
}

for key, value in checks.items():
    print(f"{key}={value}")

print(f"health_status={health_status}")
print(f"hosts_status={hosts_status}")
PY

# v1.0.3.75 CMDB quality follow-up debug field commands

用途：確認「資料品質仍未解決」時，到底是版本沒更新、API 沒回新欄位、CMDB 分類規則沒有抓到，還是資料本身缺少 OS / host_type / status。

本指令已去敏化：

- 不輸出資產名稱
- 不輸出主機名稱
- 不輸出 IP
- 不輸出 owner / 人員
- 不輸出 system_id / asset_seq 原文
- 只輸出數量、狀態、平台分布、問題類型與短 hash

## 1. 去敏化查核

在 webitgpt 主機執行：

```bash
python3 - <<'PY'
import hashlib
import json
import urllib.request

BASE = "http://127.0.0.1:8002"


def h(value):
    text = str(value or "")
    if not text:
        return ""
    return hashlib.sha256(text.encode("utf-8", "ignore")).hexdigest()[:12]


def get_json(path):
    with urllib.request.urlopen(BASE + path, timeout=15) as res:
        return json.loads(res.read().decode("utf-8"))


summary = {}

health = get_json("/health")
summary["health"] = {
    "version": health.get("version"),
    "patch_id": health.get("patch_id"),
    "status": health.get("status")
}

quality = get_json("/api/reports/data-quality")
summary["quality"] = {
    "status": quality.get("status"),
    "status_label": quality.get("status_label"),
    "score": quality.get("score"),
    "summary": quality.get("summary"),
    "breakdown": quality.get("cmdb_breakdown"),
    "detail_count": len(quality.get("cmdb_detail_rows") or [])
}

details = quality.get("cmdb_detail_rows") or []
summary["quality_detail_type_counts"] = {}
summary["quality_detail_severity_counts"] = {}
summary["platform_mismatch_samples_masked"] = []

for row in details:
    row_type = row.get("type") or "unknown"
    severity = row.get("severity") or "unknown"
    summary["quality_detail_type_counts"][row_type] = summary["quality_detail_type_counts"].get(row_type, 0) + 1
    summary["quality_detail_severity_counts"][severity] = summary["quality_detail_severity_counts"].get(severity, 0) + 1
    if row_type == "platform_mismatch" and len(summary["platform_mismatch_samples_masked"]) < 10:
        summary["platform_mismatch_samples_masked"].append({
            "asset_hash": h(row.get("asset_name")),
            "host_hash": h(row.get("hostname")),
            "ip_hash": h(row.get("ip")),
            "status": row.get("status"),
            "current_host_type": row.get("current_host_type"),
            "suggested_host_type": row.get("suggested_host_type")
        })

print(json.dumps(summary, ensure_ascii=False, indent=2, sort_keys=True))
PY
```

## 2. 若第 1 段看不出原因，再查 Mongo 去敏統計

這段只輸出統計與 hash，不輸出原文。

```bash
cd /opt/webitgpt
INSPECTION_HOME=/opt/webitgpt ./venv/bin/python - <<'PY'
import hashlib
import json
from collections import Counter
from webapp.services.mongo_service import get_collection
from webapp.services.host_service import platform_suggestion_for_host


def h(value):
    text = str(value or "")
    if not text:
        return ""
    return hashlib.sha256(text.encode("utf-8", "ignore")).hexdigest()[:12]


hosts = list(get_collection("hosts").find({}, {
    "_id": 0,
    "asset_seq": 1,
    "asset_name": 1,
    "hostname": 1,
    "ip": 1,
    "status": 1,
    "host_type": 1,
    "os": 1,
    "device_type": 1,
    "system_name": 1
}))

status_counts = Counter(str(x.get("status") or "unknown") for x in hosts)
host_type_counts = Counter(str(x.get("host_type") or "unknown") for x in hosts)
os_keyword_counts = Counter()
suggested_counts = Counter()
platform_mismatch = []

for host in hosts:
    os_text = str(host.get("os") or "").lower()
    for key in ["linux", "red hat", "rhel", "centos", "rocky", "debian", "ubuntu", "aix", "windows", "esxi"]:
        if key in os_text:
            os_keyword_counts[key] += 1
    suggestion = platform_suggestion_for_host(host)
    if suggestion.get("suggested"):
        suggested_counts[suggestion.get("suggested")] += 1
    if suggestion.get("needed"):
        platform_mismatch.append({
            "asset_hash": h(host.get("asset_name")),
            "host_hash": h(host.get("hostname")),
            "ip_hash": h(host.get("ip")),
            "status": host.get("status"),
            "current_host_type": suggestion.get("current"),
            "suggested_host_type": suggestion.get("suggested"),
            "os_hash": h(host.get("os"))
        })

result = {
    "hosts_total": len(hosts),
    "status_counts": dict(status_counts),
    "host_type_counts": dict(host_type_counts),
    "os_keyword_counts": dict(os_keyword_counts),
    "suggested_type_counts_from_os": dict(suggested_counts),
    "platform_mismatch_count": len(platform_mismatch),
    "platform_mismatch_samples_masked": platform_mismatch[:20]
}

print(json.dumps(result, ensure_ascii=False, indent=2, sort_keys=True))
PY
```

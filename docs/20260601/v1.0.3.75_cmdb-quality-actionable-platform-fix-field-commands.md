    health = json.loads(res.read().decode("utf-8"))

summary["quality_status"] = quality.get("status")
summary["cmdb_issues"] = int((quality.get("summary") or {}).get("cmdb_issues") or 0)

summary["cmdb_breakdown_count"] = len(quality.get("cmdb_breakdown") or [])
summary["cmdb_detail_count"] = len(quality.get("cmdb_detail_rows") or [])

with urllib.request.urlopen(base + "/reports/data-quality", timeout=10) as res:
    summary["data_quality_page_http"] = res.status

PY

# v1.0.3.89 核心系統影響圖主機數驗證指令

目的：證明「同一套系統的候選主機數」與「核心系統影響圖實際畫出的主機數」是否一致。

注意：本指令只輸出去敏化摘要，不會列出原始 IP、Hostname、資產編號、人名、系統名稱或 owner。

## 使用方式

在 webitgpt 主機執行。`TARGET_SYSTEM` 請在本機填入要查的系統名稱或資產名稱關鍵字，例如你畫面上看到的那套系統名稱。

```bash
cd /opt/webitgpt
OUT="/tmp/webitgpt_topology_host_count_proof_$(date +%Y%m%d%H%M%S).json"
TARGET_SYSTEM='請在本機填入要查的系統或資產名稱' OUT="$OUT" PYTHONPATH=/opt/webitgpt ./venv/bin/python - <<'PY'
import hashlib
import json
import os
import urllib.parse
import urllib.request

from webapp.services.dependency_service import list_systems, topology
from webapp.services.mongo_service import get_collection
from webapp.services.system_alias_service import host_matches_system


def digest(value):
    text = str(value or '').strip()
    if not text or text == '-':
        return ''
    return hashlib.sha256(text.encode('utf-8', 'ignore')).hexdigest()[:12]


def names_for_system(system):
    metadata = system.get('metadata') or {}
    names = [
        system.get('display_name'),
        system.get('system_id'),
        metadata.get('asset_name'),
        metadata.get('system_name'),
    ]
    return [str(item).strip() for item in names if str(item or '').strip()]


def get_json(path):
    with urllib.request.urlopen('http://127.0.0.1:8002' + path, timeout=20) as res:
        return json.loads(res.read().decode('utf-8'))


target = os.environ.get('TARGET_SYSTEM', '').strip()
if not target or target == '請在本機填入要查的系統或資產名稱':
    raise SystemExit('請先設定 TARGET_SYSTEM。')

systems = list_systems()
matched_systems = []
for system in systems:
    candidates = names_for_system(system)
    if any(target in item or item in target for item in candidates):
        matched_systems.append(system)

if not matched_systems:
    raise SystemExit('找不到符合的系統；請改用更完整的資產名稱或系統名稱。')

system = matched_systems[0]
system_id = system.get('system_id')
system_names = names_for_system(system)

hosts = list(get_collection('hosts').find({}, {
    '_id': 0,
    'asset_seq': 1,
    'asset_name': 1,
    'system_name': 1,
    'device_type': 1,
    'hostname': 1,
    'ip': 1,
    'status': 1,
    'host_type': 1,
}))

candidate_hosts = []
for host in hosts:
    if any(host_matches_system(host, name) for name in system_names):
        candidate_hosts.append(host)

api_path = '/api/dependencies/topology?' + urllib.parse.urlencode({
    'view': 'core_impact',
    'center': system_id,
    'include_unmanaged': 'false',
    'include_external': 'false',
})
api_data = get_json(api_path)
drawn_hosts = [node for node in api_data.get('nodes') or [] if str(node.get('id') or '').startswith('host:')]

candidate_by_status = {}
candidate_by_platform = {}
for host in candidate_hosts:
    candidate_by_status[str(host.get('status') or 'unknown')] = candidate_by_status.get(str(host.get('status') or 'unknown'), 0) + 1
    candidate_by_platform[str(host.get('host_type') or 'unknown')] = candidate_by_platform.get(str(host.get('host_type') or 'unknown'), 0) + 1

summary = {
    'version': get_json('/health').get('version'),
    'patch_id': get_json('/health').get('patch_id'),
    'target_hash': digest(target),
    'matched_system_count': len(matched_systems),
    'selected_system_hash': digest(system_id),
    'candidate_hosts_by_alias': len(candidate_hosts),
    'candidate_hosts_by_status': candidate_by_status,
    'candidate_hosts_by_platform': candidate_by_platform,
    'topology_drawn_hosts': len(drawn_hosts),
    'topology_meta_host_count': (api_data.get('meta') or {}).get('impact_panel', {}).get('host_count'),
    'is_fixed_when_counts_match': len(candidate_hosts) == len(drawn_hosts),
    'note': 'candidate_hosts_by_alias 應等於 topology_drawn_hosts；若不等，代表仍有拓撲補畫邏輯或資料治理問題。',
}

out = os.environ.get('OUT') or '/tmp/webitgpt_topology_host_count_proof.json'
with open(out, 'w', encoding='utf-8') as fh:
    json.dump(summary, fh, ensure_ascii=False, indent=2, sort_keys=True)
print(json.dumps({'output': out, 'sha256': hashlib.sha256(open(out, 'rb').read()).hexdigest()}, ensure_ascii=False, indent=2))
PY
cat "$OUT"
```

## 判讀方式

- `candidate_hosts_by_alias`：依系統別名 / 資產名稱 / 設備名稱比對到的主機數。
- `topology_drawn_hosts`：核心系統影響圖實際畫出的主機數。
- `is_fixed_when_counts_match=true`：代表 v1.0.3.89 的補畫邏輯在該資料上已生效。
- 若還是 `false`，請回傳這份 JSON 摘要即可，不要回傳原始主機清單。

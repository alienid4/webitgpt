# v1.0.3.63 現場查詢指令：資產與拓撲主機數不一致去敏版

用途：收集「資產管理看到的主機數」與「核心影響圖畫出的主機數」差異，但不輸出人名、主機名、IP、資產編號、owner、系統代號或內部路徑等敏感資訊。

## 使用方式

請在目標 webitgpt 主機上執行。執行前由現場人員自行填入查詢條件，輸出會自動去敏。

```bash
cat >/tmp/webitgpt_asset_topology_sanitized_check.sh <<'SH'
#!/usr/bin/env bash
set -u

STAMP="$(date +%Y%m%d%H%M%S)"
OUT="/tmp/webitgpt_asset_topology_sanitized_${STAMP}.txt"
exec > >(tee "${OUT}") 2>&1

read -r -p "請輸入查詢關鍵字，不會原文輸出: " SEARCH_TERM
read -r -p "請輸入拓撲中心系統代號，不會原文輸出: " CENTER_SYSTEM_ID

echo "==== BASIC SANITIZED ===="
date '+%Y-%m-%d %H:%M:%S %z'
echo "host_hash=$(hostname -f 2>/dev/null | sha256sum | awk '{print substr($1,1,12)}')"
cat /etc/redhat-release 2>/dev/null | sed -E 's/[A-Za-z0-9_.-]+/[redacted]/g' || true

echo
echo "==== SERVICE SANITIZED ===="
systemctl is-active webitgpt 2>/dev/null | sed 's/.*/webitgpt_service=&/' || true
ss -ltn 2>/dev/null | awk '/:8002|:27017/ {count++} END {print "listening_8002_27017_count=" count+0}'

echo
echo "==== HEALTH SANITIZED ===="
curl -sS http://127.0.0.1:8002/health 2>/dev/null \
  | sed -E 's/"build_time":"[^"]+"/"build_time":"[redacted]"/g; s/"release_note":"[^"]+"/"release_note":"[redacted]"/g' || true
echo

echo
echo "==== DATA SANITIZED ===="
cd /opt/webitgpt || exit 1
PYBIN="/opt/webitgpt/venv/bin/python"
if [ ! -x "${PYBIN}" ]; then
  PYBIN="python3"
fi

WEBITGPT_SEARCH_TERM="${SEARCH_TERM}" WEBITGPT_CENTER_SYSTEM_ID="${CENTER_SYSTEM_ID}" "${PYBIN}" - <<'PY'
import hashlib
import json
import os

from webapp import config
from webapp.services.mongo_service import get_db
from webapp.services import dependency_service

db = get_db()
search_term = os.environ.get("WEBITGPT_SEARCH_TERM", "")
center_system_id = os.environ.get("WEBITGPT_CENTER_SYSTEM_ID", "")

def h(value):
    text = str(value or "").strip()
    if not text:
        return ""
    return hashlib.sha256(text.encode("utf-8")).hexdigest()[:12]

def scrub_doc(doc, fields):
    return {field + "_hash": h(doc.get(field)) for field in fields if doc.get(field)}

def host_key(doc):
    return str(doc.get("asset_seq") or doc.get("hostname") or doc.get("ip") or "").strip()

host_fields = {
    "_id": 0,
    "asset_seq": 1,
    "asset_name": 1,
    "system_name": 1,
    "hostname": 1,
    "ip": 1,
    "status": 1,
    "host_type": 1,
    "device_type": 1,
    "group_name": 1,
    "apid": 1,
}
system_fields = {
    "_id": 0,
    "system_id": 1,
    "display_name": 1,
    "host_refs": 1,
    "metadata": 1,
    "owner": 1,
}

def text_hit(doc):
    keys = ["asset_seq", "asset_name", "system_name", "hostname", "ip", "group_name", "apid"]
    blob = " ".join(str(doc.get(key) or "") for key in keys)
    return bool(search_term and search_term in blob)

print("CONFIG_SANITIZED")
print(json.dumps({
    "version": config.VERSION,
    "patch_id": config.PATCH_ID,
    "search_term_hash": h(search_term),
    "center_system_hash": h(center_system_id),
}, ensure_ascii=False))

hosts = list(db.hosts.find({}, host_fields))
matched_hosts = [host for host in hosts if text_hit(host)]
matched_host_keys = {host_key(host) for host in matched_hosts if host_key(host)}

print("\nHOST_COUNTS")
print(json.dumps({
    "hosts_total": len(hosts),
    "matched_hosts": len(matched_hosts),
    "matched_host_key_hashes": sorted(h(item) for item in matched_host_keys),
}, ensure_ascii=False))

print("\nMATCHED_HOSTS_SANITIZED")
for host in matched_hosts:
    payload = {
        "host_key_hash": h(host_key(host)),
        "status": host.get("status") or "",
        "host_type_hash": h(host.get("host_type")),
        "device_type_hash": h(host.get("device_type")),
    }
    payload.update(scrub_doc(host, ["asset_name", "system_name", "hostname", "ip", "asset_seq", "group_name", "apid"]))
    print(json.dumps(payload, ensure_ascii=False, sort_keys=True))

systems = list(db.dependency_systems.find({}, system_fields))
related_systems = []
for system in systems:
    display = str(system.get("display_name") or "")
    refs = {str(item) for item in system.get("host_refs") or []}
    if system.get("system_id") == center_system_id or (search_term and search_term in display) or refs.intersection(matched_host_keys):
        related_systems.append(system)

print("\nDEPENDENCY_SYSTEM_COUNTS")
print(json.dumps({
    "dependency_systems_total": len(systems),
    "related_systems": len(related_systems),
    "related_system_hashes": sorted(h(item.get("system_id")) for item in related_systems),
}, ensure_ascii=False))

print("\nRELATED_SYSTEMS_SANITIZED")
for system in related_systems:
    print(json.dumps({
        "system_id_hash": h(system.get("system_id")),
        "display_name_hash": h(system.get("display_name")),
        "owner_hash": h(system.get("owner")),
        "host_ref_count": len(system.get("host_refs") or []),
        "host_ref_hashes": sorted(h(item) for item in (system.get("host_refs") or [])),
        "metadata_asset_name_hash": h((system.get("metadata") or {}).get("asset_name")),
        "metadata_system_name_hash": h((system.get("metadata") or {}).get("system_name")),
    }, ensure_ascii=False, sort_keys=True))

print("\nTOPOLOGY_SANITIZED")
topology = dependency_service.topology(view="core_impact", center=center_system_id, limit=200)
nodes = topology.get("nodes") or []
edges = topology.get("edges") or []
host_nodes = [node for node in nodes if str(node.get("id") or "").startswith("host:")]
system_nodes = [node for node in nodes if not str(node.get("id") or "").startswith(("host:", "core:"))]
drawn_keys = {str(node.get("id") or "").replace("host:", "", 1) for node in host_nodes}

print(json.dumps({
    "meta_system_count": ((topology.get("meta") or {}).get("impact_panel") or {}).get("system_count"),
    "meta_host_count": ((topology.get("meta") or {}).get("impact_panel") or {}).get("host_count"),
    "node_count": len(nodes),
    "system_node_count": len(system_nodes),
    "host_node_count": len(host_nodes),
    "edge_count": len(edges),
    "system_node_hashes": sorted(h(node.get("id")) for node in system_nodes),
    "drawn_host_key_hashes": sorted(h(item) for item in drawn_keys),
}, ensure_ascii=False, sort_keys=True))

print("\nCOMPARE_SANITIZED")
print(json.dumps({
    "matched_host_count": len(matched_host_keys),
    "drawn_host_count": len(drawn_keys),
    "missing_in_topology_hashes": sorted(h(item) for item in (matched_host_keys - drawn_keys)),
    "extra_in_topology_hashes": sorted(h(item) for item in (drawn_keys - matched_host_keys)),
}, ensure_ascii=False, sort_keys=True))
PY

echo
echo "==== PAGE SMOKE SANITIZED ===="
HTTP_CODE="$(curl -sS -o /tmp/webitgpt_core_impact_${STAMP}.html -w "%{http_code}" "http://127.0.0.1:8002/dependencies?view=core_impact&center=${CENTER_SYSTEM_ID}" 2>/dev/null || true)"
echo "core_impact_http=${HTTP_CODE}"
grep -o 'host:[^"]*' /tmp/webitgpt_core_impact_${STAMP}.html 2>/dev/null \
  | sha256sum \
  | awk '{print "rendered_host_marker_hash=" substr($1,1,12)}' || true
rm -f /tmp/webitgpt_core_impact_${STAMP}.html

echo
echo "==== OUTPUT FILE ===="
echo "${OUT}"
SH

bash /tmp/webitgpt_asset_topology_sanitized_check.sh
```

## 回傳方式

回傳最後產生的 `.txt` 即可，例如：

```bash
cat /tmp/webitgpt_asset_topology_sanitized_YYYYMMDDHHMMSS.txt
```

## 去敏原則

- 不輸出原始搜尋字串。
- 不輸出原始 IP、hostname、asset_seq、asset_name、system_name、owner、APID。
- 不輸出主機完整 FQDN。
- 不輸出 systemd 詳細 log。
- 使用 12 碼 SHA-256 hash 做一致比對，方便判斷「哪一筆漏掉」，但不能從輸出還原原始值。
